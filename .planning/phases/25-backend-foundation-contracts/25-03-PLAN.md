```ts
export const MAX_BODY_BYTES = 65536;      // 64 KB (D-08)

export const MAX_SEATS = 8;

export const QuestionIdSchema;            // /^[a-z0-9-]+$/, max 100
export const SeatSchema;                  // strict { seat 0..7, packIds uuid[] 0..20, comboId uuid|null, difficulty easy|medium|hard|null }
export const OutcomeSchema;               // strict { seq int 0..9999, seat 0..7, packId uuid, questionId, contentHash, category, difficulty|null, correct boolean }
export const GameStatusSchema;            // enum finished | abandoned
export const GameRecordSchema;            // strict { schemaVersion literal 1, gameId uuid, installId uuid, status, startedAt iso, endedAt iso,
                                          //   playerCount int 1..8, winnerSeat int 0..7 | null, championshipReached boolean,
                                          //   seats SeatSchema[] 1..8, outcomes OutcomeSchema[] 0..500 } + superRefine (below)
export const FeedbackReasonSchema;        // enum wrong_answer | ambiguous | outdated | typo | offensive | other
export const QuestionSnapshotSchema;      // strict { questionText 1..500, answerText 1..200, choices string(max 200)[] max 6 optional, category, difficulty|null optional }
export const FeedbackSchema;              // strict { schemaVersion literal 1, feedbackId uuid, installId uuid, gameId uuid|null optional,
                                          //   packId uuid, questionId, contentHash, reason, note string max 500 optional,
                                          //   questionSnapshot, createdAt iso } + superRefine (reason other => trimmed note non-empty)
export const IngestReceiptSchema;         // strict { id uuid, receivedAt iso }
export const ApiErrorCodeSchema;          // enum invalid_request | payload_too_large | rate_limited | method_not_allowed | not_found | backend_disabled | internal_error
export const ApiErrorSchema;              // strict { error ApiErrorCode, issues?: { path string, message string }[] max 20 }
export const HealthResponseSchema;        // strict { status ok|degraded, db: { status ok|down, latencyMs number|null }, time iso }
export type GameRecord, Outcome, Seat, Feedback, FeedbackReason, QuestionSnapshot, IngestReceipt, ApiError, ApiErrorCode, HealthResponse = z.infer<...>;
```
GameRecordSchema superRefine rules: seats.length === playerCount; seat values unique and each < playerCount; every outcome.seat < playerCount; outcome seq values unique; winnerSeat (when non-null) < playerCount; status finished requires winnerSeat non-null; Date.parse(endedAt) >= Date.parse(startedAt).

packages/api/src/repo.ts
```ts
import type { GameRecord, Feedback } from '@trivial-world/types';
export interface InsertResult { id: string; receivedAt: string; created: boolean }   // receivedAt = ISO string from DB
export interface IngestRepository {
  /** One transaction: game + seats + outcomes. ON CONFLICT (id) DO NOTHING; replay returns stored receivedAt with created=false. */
  insertGameRecord(record: GameRecord): Promise<InsertResult>;
  /** ON CONFLICT DO NOTHING on id or on (install_id, pack_id, question_id, reason); conflict returns the existing row's id/receivedAt with created=false. */
  insertFeedback(feedback: Feedback): Promise<InsertResult>;
  /** Resolves if the DB answers SELECT 1; rejects otherwise. */
  ping(): Promise<void>;
}
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| anonymous client -> API | Every field of the request body is attacker-controlled; these schemas are the server's input filter |
| LLM/script -> review API (Phase 30) | Review decisions and notes are untrusted |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-25-10 | Tampering | GameRecordSchema / FeedbackSchema | mitigate | z.strictObject everywhere (unknown keys rejected), z.uuid strict, regex for questionId/contentHash, array caps (500 outcomes, 8 seats, 20 packIds), string caps; tests cover each |
| T-25-11 | Information disclosure | Player names reaching server | mitigate | No name field exists; strict schema rejects any extra key; test asserts `players`/`name` rejected (D-10, REC-05) |
| T-25-12 | Denial of service | Large nested payloads | mitigate | Caps on every array and string; body byte cap MAX_BODY_BYTES consumed by 25-04 before parse |
| T-25-13 | Tampering | Review notes as prompt-injection vector | mitigate | Field named untrustedNote, max 500, so Phase 30 must delimit it as data |
| T-25-14 | Repudiation | Client-supplied timestamps | accept | Client times stored as client claims; server received_at is authoritative (D-10, REC-06) |
</threat_model>
